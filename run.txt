def save_calibration(corners):
    with open(CALIB_FILE, 'w') as f:
        json.dump(corners, f)
    print(f"💾 کالیبراسیون ذخیره شد در {CALIB_FILE}")

def load_calibration():
    if os.path.exists(CALIB_FILE):
        with open(CALIB_FILE, 'r') as f:
            corners = json.load(f)
        print(f"✅ کالیبراسیون بارگذاری شد از {CALIB_FILE}")
        return corners
    return None

def get_perspective_matrix(corners):
    src_points = np.array(corners, dtype=np.float32)
    dst_points = np.array([
        [0, 0],
        [SCREEN_W, 0],
        [SCREEN_W, SCREEN_H],
        [0, SCREEN_H]
    ], dtype=np.float32)
    matrix = cv2.getPerspectiveTransform(src_points, dst_points)
    return matrix

def transform_point(matrix, point):
    point = np.array([point], dtype=np.float32).reshape(-1, 1, 2)
    transformed = cv2.perspectiveTransform(point, matrix)
    return int(transformed[0][0][0]), int(transformed[0][0][1])

def get_finger_state(hand_landmarks):
    fingers = []

    thumb_tip = hand_landmarks[4]
    thumb_ip = hand_landmarks[3]
    if thumb_tip.x < thumb_ip.x:
        fingers.append(1)
    else:
        fingers.append(0)

    finger_tips = [8, 12, 16, 20]
    finger_pips = [6, 10, 14, 18]

    for tip_idx, pip_idx in zip(finger_tips, finger_pips):
        tip = hand_landmarks[tip_idx]
        pip = hand_landmarks[pip_idx]
        if tip.y < pip.y:
            fingers.append(1)
        else:
            fingers.append(0)

    return fingers

def calculate_hand_size(hand_landmarks):
    """محاسبه اندازه دست نسبت به تصویر"""
    # فاصله بین مچ (0) و نوک انگشت وسط (12)
    wrist = hand_landmarks[0]
    middle_tip = hand_landmarks[12]

    distance = np.sqrt(
        (wrist.x - middle_tip.x)**2 +
        (wrist.y - middle_tip.y)**2
    )
    return distance

def apply_zoom(frame, hand_landmarks, zoom_factor):
    """اعمال زوم روی ناحیه دست"""
    if not zoom_enabled or zoom_factor <= 1.0:
        return frame

    # مرکز دست (مچ)
    wrist_x = int(hand_landmarks[0].x * FRAME_W)
    wrist_y = int(hand_landmarks[0].y * FRAME_H)

    # محاسبه ناحیه زوم
    h, w = frame.shape[:2]
    zoom_w = int(w / zoom_factor)
    zoom_h = int(h / zoom_factor)

    x1 = max(0, wrist_x - zoom_w // 2)
    y1 = max(0, wrist_y - zoom_h // 2)
    x2 = min(w, x1 + zoom_w)
    y2 = min(h, y1 + zoom_h)

    # برش و تغییر اندازه
    cropped = frame[y1:y2, x1:x2]
    zoomed = cv2.resize(cropped, (w, h), interpolation=cv2.INTER_LINEAR)

    return zoomed
